cmake_minimum_required(VERSION 3.16)
project(m10_rknn_probe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
if(NOT DEFINED M10_RKNPU_DIR)
  message(FATAL_ERROR "Set M10_RKNPU_DIR to the prepared official RKNPU2 v1.0.0 assets")
endif()
if(NOT DEFINED M10_RKNNRT_LIB)
  message(FATAL_ERROR "Set M10_RKNNRT_LIB to librknnrt.so pulled from the target device")
endif()
add_library(rknnrt SHARED IMPORTED)
set_target_properties(rknnrt PROPERTIES IMPORTED_LOCATION "${M10_RKNNRT_LIB}")
add_executable(m10_rknn_probe rknn_probe.cpp "${M10_RKNPU_DIR}/ssd/ssd.cc")
target_include_directories(m10_rknn_probe PRIVATE "${M10_RKNPU_DIR}/include" "${M10_RKNPU_DIR}/ssd")
target_link_libraries(m10_rknn_probe PRIVATE rknnrt log)
install(TARGETS m10_rknn_probe RUNTIME DESTINATION bin)
